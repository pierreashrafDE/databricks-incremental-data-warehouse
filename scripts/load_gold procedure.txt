CREATE OR REPLACE PROCEDURE `warehouse-project`.gold.load_gold()
LANGUAGE SQL
SQL SECURITY INVOKER
MODIFIES SQL DATA
AS
BEGIN

    -- =========================================================
    -- START GOLD LAYER
    -- =========================================================

    SELECT '========================================' AS message;
    SELECT 'STARTING GOLD LAYER' AS message;
    SELECT '========================================' AS message;


    -- =========================================================
    -- DIMENSION: CUSTOMERS
    -- =========================================================

    SELECT 'Creating gold.dim_customers...' AS message;

    CREATE OR REPLACE VIEW `warehouse-project`.gold.dim_customers AS

    SELECT
        ROW_NUMBER() OVER (
            ORDER BY ci.cst_id
        ) AS customer_key,

        ci.cst_id AS customer_id,
        ci.cst_key AS customer_number,
        ci.cst_firstname AS first_name,
        ci.cst_lastname AS last_name,

        la.CNTRY AS country,

        ci.cst_marital_status AS marital_status,

        CASE
            WHEN ci.cst_gndr <> 'N/A'
            THEN ci.cst_gndr

            ELSE COALESCE(ca.GEN, 'N/A')
        END AS gender,

        ca.BDATE AS birthdate,

        ci.cst_create_date AS create_date

    FROM `warehouse-project`.silver.crm_customers ci

    LEFT JOIN `warehouse-project`.silver.erp_cust_az12 ca
        ON ci.cst_key = ca.CID

    LEFT JOIN `warehouse-project`.silver.erp_location la
        ON ci.cst_key = la.CID;


    -- =========================================================
    -- DIMENSION: PRODUCTS
    -- =========================================================

    SELECT 'Creating gold.dim_products...' AS message;

    CREATE OR REPLACE VIEW `warehouse-project`.gold.dim_products AS

    SELECT
        ROW_NUMBER() OVER (
            ORDER BY pn.prd_start_dt, pn.prd_key
        ) AS product_key,

        pn.prd_id AS product_id,
        pn.prd_key AS product_number,
        pn.prd_nm AS product_name,

        pn.cat_id AS category_id,

        pc.CAT AS category,
        pc.SUBCAT AS subcategory,
        pc.MAINTENANCE AS maintenance,

        pn.prd_cost AS cost,
        pn.prd_line AS product_line,
        pn.prd_start_dt AS start_date

    FROM `warehouse-project`.silver.crm_products pn

    LEFT JOIN `warehouse-project`.silver.erp_prd_cat pc
        ON pn.cat_id = pc.ID

    WHERE pn.prd_end_dt IS NULL;


    -- =========================================================
    -- FACT: SALES
    -- =========================================================

    SELECT 'Creating gold.fact_sales...' AS message;

    CREATE OR REPLACE VIEW `warehouse-project`.gold.fact_sales AS

    SELECT

        sd.sls_ord_num AS order_number,

        pr.product_key AS product_key,

        cu.customer_key AS customer_key,

        sd.sls_order_dt AS order_date,
        sd.sls_ship_dt AS shipping_date,
        sd.sls_due_dt AS due_date,

        sd.sls_sales AS sales_amount,
        sd.sls_quantity AS quantity,
        sd.sls_price AS price

    FROM `warehouse-project`.silver.crm_sales sd

    LEFT JOIN `warehouse-project`.gold.dim_products pr
        ON sd.sls_prd_key = pr.product_number

    LEFT JOIN `warehouse-project`.gold.dim_customers cu
        ON sd.sls_cust_id = cu.customer_id;


    -- =========================================================
    -- COMPLETE
    -- =========================================================

    SELECT '========================================' AS message;
    SELECT 'GOLD LAYER COMPLETED SUCCESSFULLY' AS message;
    SELECT '========================================' AS message;

END;
